In src/App.jsx

import React from "react";
import "./App.css";
function App()
{
    const num1=20;
    const num2=30;
    function calculateSum(a,b){
        return a+b;
    }
    return(
        <div>
         <h1>React-Sum of two numbers</h1>
         <p>First Number:{num1}</p>
         <p>Second Number:{num2}</p>
         <h2>Sum={calculateSum(num1,num2)}</h2>
        </div>
    );
}
export default App;
