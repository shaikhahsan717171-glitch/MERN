In src/Counter.jsx


import { useState } from "react";

function Counter(){

    const [count,setCount]=useState(0);

    return(

         <div>

           <h3>Counter using State</h3>



            <h2>{count}</h2>

            <button onClick={()=>setCount(count+1)}></button>

            <button onClick={()=>setCount(count-1)}></button>

            <hr/>

          </div>

     );

}

export default Counter;
